4. MODELO DE BASE DE DATOS (CLOUD SQL - MARIADB)
--------------------------------------------------------------------------------

-- MÓDULO DE SEGURIDAD (RF01-RF04)
CREATE TABLE roles (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,  -- admin, docente, estudiante, consultor
    descripcion TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE usuarios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    rol_id INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    ultimo_acceso TIMESTAMP NULL,
    token_recuperacion VARCHAR(255) NULL,
    fecha_expiracion_token TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (rol_id) REFERENCES roles(id)
);

-- MÓDULO DE FINCAS (RF05-RF08)
CREATE TABLE departamentos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,
    codigo_dane VARCHAR(10)
);

CREATE TABLE municipios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    departamento_id INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    codigo_dane VARCHAR(10),
    FOREIGN KEY (departamento_id) REFERENCES departamentos(id)
);

CREATE TABLE sistemas_productivos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,  -- familiar, extensivo, semi_intensivo, intensivo
    descripcion TEXT
);

CREATE TABLE fincas (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(200) NOT NULL,
    propietario VARCHAR(200) NOT NULL,
    departamento_id INT NOT NULL,
    municipio_id INT NOT NULL,
    vereda VARCHAR(100),
    direccion VARCHAR(255),
    latitud DECIMAL(10,8) NULL,
    longitud DECIMAL(11,8) NULL,
    sistema_productivo_id INT NOT NULL,
    usuario_id INT NOT NULL,  -- propietario/encargado
    activo BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (departamento_id) REFERENCES departamentos(id),
    FOREIGN KEY (municipio_id) REFERENCES municipios(id),
    FOREIGN KEY (sistema_productivo_id) REFERENCES sistemas_productivos(id),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE especies (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(100) NOT NULL,  -- ovino, caprino, porcino, etc.
    nombre_cientifico VARCHAR(100),
    activo BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE finca_especie (
    id INT PRIMARY KEY AUTO_INCREMENT,
    finca_id INT NOT NULL,
    especie_id INT NOT NULL,
    FOREIGN KEY (finca_id) REFERENCES fincas(id),
    FOREIGN KEY (especie_id) REFERENCES especies(id),
    UNIQUE KEY unique_finca_especie (finca_id, especie_id)
);

CREATE TABLE inventario_animal (
    id INT PRIMARY KEY AUTO_INCREMENT,
    finca_id INT NOT NULL,
    especie_id INT NOT NULL,
    grupo_etario VARCHAR(50) NOT NULL,  -- lactantes, jovenes, adultos
    categoria VARCHAR(50),  -- machos, hembras, reproductores
    cantidad INT NOT NULL,
    fecha_registro DATE NOT NULL,
    FOREIGN KEY (finca_id) REFERENCES fincas(id),
    FOREIGN KEY (especie_id) REFERENCES especies(id)
);

-- MÓDULO DE PARAMETRIZACIÓN (RF09-RF12)
CREATE TABLE versiones_parametrizacion (
    id INT PRIMARY KEY AUTO_INCREMENT,
    especie_id INT NOT NULL,
    version INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (especie_id) REFERENCES especies(id)
);

CREATE TABLE dimensiones (
    id INT PRIMARY KEY AUTO_INCREMENT,
    especie_id INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,  -- 5 libertades
    descripcion TEXT,
    peso DECIMAL(5,2) NOT NULL,  -- 30, 30, 30, 10
    orden INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (especie_id) REFERENCES especies(id)
);

CREATE TABLE criterios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    dimension_id INT NOT NULL,
    nombre VARCHAR(200) NOT NULL,
    descripcion TEXT,
    peso DECIMAL(5,2) NOT NULL,
    orden INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (dimension_id) REFERENCES dimensiones(id)
);

CREATE TABLE indicadores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    criterio_id INT NOT NULL,
    nombre VARCHAR(200) NOT NULL,
    descripcion TEXT,
    metodo_evaluacion ENUM('documental', 'visual', 'muestreo') NOT NULL,
    tipo_respuesta ENUM('si_no', 'seleccion_unica', 'seleccion_multiple', 
                        'escala', 'numerica', 'textual', 'evidencia') NOT NULL,
    umbral_aprobacion DECIMAL(10,2) NULL,
    peso DECIMAL(5,2) NOT NULL,
    orden INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (criterio_id) REFERENCES criterios(id)
);

CREATE TABLE opciones_respuesta (
    id INT PRIMARY KEY AUTO_INCREMENT,
    indicador_id INT NOT NULL,
    opcion VARCHAR(255) NOT NULL,
    valor DECIMAL(10,2) NOT NULL,
    orden INT NOT NULL,
    FOREIGN KEY (indicador_id) REFERENCES indicadores(id)
);

-- MÓDULO DE EVALUACIÓN (RF13-RF23)
CREATE TABLE estados_evaluacion (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,  -- borrador, en_proceso, finalizada, validada, cerrada
    descripcion TEXT
);

CREATE TABLE evaluaciones (
    id INT PRIMARY KEY AUTO_INCREMENT,
    finca_id INT NOT NULL,
    especie_id INT NOT NULL,
    evaluador_id INT NOT NULL,
    estado_id INT NOT NULL DEFAULT 1,
    puntaje_total DECIMAL(5,2) NULL,
    clasificacion VARCHAR(50) NULL,  -- excelente, alto, medio, bajo
    observaciones_generales TEXT NULL,
    fecha_inicio TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_finalizacion TIMESTAMP NULL,
    fecha_guardado_parcial TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (finca_id) REFERENCES fincas(id),
    FOREIGN KEY (especie_id) REFERENCES especies(id),
    FOREIGN KEY (evaluador_id) REFERENCES usuarios(id),
    FOREIGN KEY (estado_id) REFERENCES estados_evaluacion(id)
);

CREATE TABLE detalle_evaluacion (
    id INT PRIMARY KEY AUTO_INCREMENT,
    evaluacion_id INT NOT NULL,
    indicador_id INT NOT NULL,
    respuesta TEXT NULL,
    puntaje_obtenido DECIMAL(5,2) NULL,
    porcentaje_cumplimiento DECIMAL(5,2) NULL,
    observaciones TEXT NULL,
    FOREIGN KEY (evaluacion_id) REFERENCES evaluaciones(id),
    FOREIGN KEY (indicador_id) REFERENCES indicadores(id)
);

CREATE TABLE evidencias (
    id INT PRIMARY KEY AUTO_INCREMENT,
    detalle_evaluacion_id INT NOT NULL,
    tipo ENUM('imagen', 'documento', 'video') NOT NULL,
    nombre_archivo VARCHAR(255) NOT NULL,
    ruta_cloud VARCHAR(500) NOT NULL,  -- Cloud Storage URL
    tamano_bytes INT NOT NULL,
    tipo_mime VARCHAR(100) NOT NULL,
    fecha_subida TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (detalle_evaluacion_id) REFERENCES detalle_evaluacion(id)
);

-- MÓDULO DE DIAGNÓSTICO Y RECOMENDACIONES (RF22-RF27)
CREATE TABLE diagnosticos (
    id INT PRIMARY KEY AUTO_INCREMENT,
    evaluacion_id INT NOT NULL,
    fortalezas TEXT NULL,
    incumplimientos TEXT NULL,
    alertas_criticas TEXT NULL,
    oportunidades_mejora TEXT NULL,
    fecha_generacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (evaluacion_id) REFERENCES evaluaciones(id)
);

CREATE TABLE recomendaciones (
    id INT PRIMARY KEY AUTO_INCREMENT,
    evaluacion_id INT NOT NULL,
    indicador_id INT NULL,
    tipo ENUM('ia', 'manual') NOT NULL,
    descripcion TEXT NOT NULL,
    prioridad ENUM('alta', 'media', 'baja') NOT NULL,
    estado ENUM('pendiente', 'en_proceso', 'completada') DEFAULT 'pendiente',
    responsable VARCHAR(200) NULL,
    plazo DATE NULL,
    editada_por INT NULL,
    fecha_edicion TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (evaluacion_id) REFERENCES evaluaciones(id),
    FOREIGN KEY (indicador_id) REFERENCES indicadores(id),
    FOREIGN KEY (editada_por) REFERENCES usuarios(id)
);

-- MÓDULO DE CERTIFICACIÓN (RF38)
CREATE TABLE certificaciones (
    id INT PRIMARY KEY AUTO_INCREMENT,
    evaluacion_id INT NOT NULL,
    numero_certificado VARCHAR(100) UNIQUE NOT NULL,
    fecha_emision TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    fecha_vencimiento TIMESTAMP NOT NULL,
    estado ENUM('vigente', 'vencido', 'revocado') DEFAULT 'vigente',
    ruta_pdf VARCHAR(500) NULL,
    enviado_por_correo BOOLEAN DEFAULT FALSE,
    fecha_envio TIMESTAMP NULL,
    FOREIGN KEY (evaluacion_id) REFERENCES evaluaciones(id)
);

-- MÓDULO DE AUDITORÍA (RF34, RNF08)
CREATE TABLE auditoria (
    id INT PRIMARY KEY AUTO_INCREMENT,

    datos_nuevos JSON NULL,

    user_agent VARCHAR(500) NULL,
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

-- MÓDULO DE NOTIFICACIONES (RF40)
CREATE TABLE notificaciones (

-- PROCEDIMIENTOS ALMACENADOS
DELIMITER //

CREATE PROCEDURE sp_calcular_puntaje_evaluacion(
    IN p_evaluacion_id INT
)
BEGIN
    DECLARE v_puntaje_total DECIMAL(5,2);
    
    SELECT SUM(de.puntaje_obtenido * i.peso / 100)
    INTO v_puntaje_total
    FROM detalle_evaluacion de
    JOIN indicadores i ON de.indicador_id = i.id
    WHERE de.evaluacion_id = p_evaluacion_id;
    
    UPDATE evaluaciones
    SET puntaje_total = v_puntaje_total,
        clasificacion = CASE
            WHEN v_puntaje_total >= 90 THEN 'excelente'
            WHEN v_puntaje_total >= 75 THEN 'alto'
            WHEN v_puntaje_total >= 50 THEN 'medio'
            ELSE 'bajo'
        END
    WHERE id = p_evaluacion_id;
END //

DELIMITER ;


5. CONFIGURACIÓN GOOGLE CLOUD (CLOUDBUILD.YAML)
--------------------------------------------------------------------------------

steps:
  # Build del backend
  - name: 'node:18'
    entrypoint: 'npm'
    args: ['ci']
    dir: 'backend'
  
  - name: 'node:18'
    entrypoint: 'npm'
    args: ['run', 'build']
    dir: 'backend'
  
  # Test
  - name: 'node:18'
    entrypoint: 'npm'
    args: ['run', 'test']
    dir: 'backend'
  
  # Deploy backend a Cloud Run
  - name: 'gcr.io/google.com/cloudsdktool/cloud-sdk'
    entrypoint: gcloud
    args:
      - 'run'
      - 'deploy'
      - 'ica-backend'
      - '--source=./backend'
      - '--region=us-central1'
      - '--platform=managed'
      - '--allow-unauthenticated'
  
  # Build del frontend
  - name: 'node:18'
    entrypoint: 'npm'
    args: ['ci']
    dir: 'frontend'
  
  - name: 'node:18'
    entrypoint: 'npm'
    args: ['run', 'build']
    dir: 'frontend'
  
  # Deploy frontend a Firebase Hosting
  - name: 'gcr.io/google.com/cloudsdktool/cloud-sdk'
    entrypoint: 'firebase'
    args: ['deploy', '--only', 'hosting']
